# Python Code Generated

```python
import random
import time

class Machine:
    def __init__(self, machine_id):
        self.machine_id = machine_id
        self.temperature = 0
        self.vibration = 0
        self.energy = 0

    def generate_sensor_data(self):
        """Simulate IoT sensor data"""
        self.temperature = random.randint(60, 120)
        self.vibration = random.randint(10, 80)
        self.energy = random.randint(200, 500)

    def analyze_status(self):
        """Analyze machine condition"""
        if self.temperature > 100 or self.vibration > 60:
            return "Maintenance Required"
        elif self.temperature > 90:
            return "Warning: Check Machine"
        else:
            return "Machine Operating Normally"

    def display_report(self):
        """Display machine status"""
        print("\nMachine ID:", self.machine_id)
        print("Temperature:", self.temperature, "°C")
        print("Vibration Level:", self.vibration)
        print("Energy Consumption:", self.energy, "Watts")
        print("Status:", self.analyze_status())


def monitor_factory():
    """Simulate monitoring multiple machines"""
    machines = []

    # Create machines
    for i in range(1, 6):
        machines.append(Machine(i))

    print("------ Factory Monitoring System ------")

    # Generate sensor data and display reports
    for machine in machines:
        machine.generate_sensor_data()
        machine.display_report()
        time.sleep(1)


# Run monitoring system
monitor_factory()
```
